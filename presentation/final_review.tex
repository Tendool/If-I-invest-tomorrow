\begin{frame}{Objectives}
  \vspace{0.3cm}
  \begin{columns}[T]
    \column{0.47\textwidth}
      \begin{enumerate}\setlength{\itemsep}{9pt}
        \item Take the investor's \textbf{amount, horizon, risk tolerance, target and sectors}
        \item \textbf{Analyse the market}: returns, risk, correlation, market regime, anomalies

      \end{enumerate}
  \end{columns}

% ------------------------------------------------------------------ 7 literature gap
\begin{frame}{Literature gap}
  \vspace{0.3cm}
  \begin{columns}[T]
    \column{0.47\textwidth}
      \eyebrow{What exists}\par\vspace{2pt}\hair\par\vspace{6pt}
      \small
      \begin{itemize}\setlength{\itemsep}{7pt}
        \item Prediction tools give a \textbf{single price forecast}, with no range of outcomes
        \item Robo-advisors give an allocation but \textbf{no probability of reaching a goal} and no crash test
        \item ML studies report \textbf{weak return predictability}, not a usable plan
        \item Finance chatbots \textbf{invent numbers} and cannot act safely
      \end{itemize}
    \column{0.47\textwidth}
      \eyebrow{What we add}\par\vspace{2pt}\hair\par\vspace{6pt}
      \small
      \begin{itemize}\setlength{\itemsep}{7pt}
        \item One system: data $\to$ models $\to$ \textbf{plan in rupees and shares}
        \item \textbf{Probabilities}, ranges and \textbf{stress tests} for every strategy
        \item Honest validation: use ML only where it shows skill (volatility)
        \item An agent \textbf{grounded in the engine}, trading only under rules checked in code
        \item Built for \textbf{Indian markets}
      \end{itemize}
  \end{columns}
\end{frame}
